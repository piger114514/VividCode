cmake_minimum_required(VERSION 3.21)
project(vividcode LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Sql Network)
add_library(vividcode_store STATIC ProjectStore.cpp ProjectStore.h ProjectScanner.h SharedRuntime.h ReadPolicy.h GenerationControl.h)
target_link_libraries(vividcode_store PUBLIC Qt6::Core)
add_executable(vividcode cli_main.cpp ConsoleOutput.h JavaScriptIndexer.cpp JavaScriptIndexer.h JavaIndexer.cpp JavaIndexer.h AnalysisEnvironment.h)
target_link_libraries(vividcode PRIVATE vividcode_store)
add_executable(vividcode-viewer WIN32 main.cpp VividCode.cpp VividCode.h
    SemanticPanel.cpp SemanticPanel.h
    AiExplanation.cpp AiExplanation.h
    PiAgent.cpp PiAgent.h PiRichText.h
    RelationGraph.cpp RelationGraph.h RelationLayout.h
    ProjectScanner.h ProjectDetection.h UnifiedGraphStore.h SourcePopup.h)
target_link_libraries(vividcode-viewer PRIVATE Qt6::Widgets Qt6::Sql Qt6::Network vividcode_store)
if(WIN32)
    target_link_libraries(vividcode-viewer PRIVATE crypt32)
endif()
if(MSVC)
    foreach(target vividcode vividcode-viewer vividcode_store)
        target_compile_options(${target} PRIVATE /utf-8 /W4)
    endforeach()
endif()
add_executable(vividcode-launcher WIN32 ProjectLauncher.cpp)
set_target_properties(vividcode-launcher PROPERTIES AUTOMOC OFF AUTORCC OFF)
target_link_libraries(vividcode-launcher PRIVATE shell32 ole32 user32 advapi32)
if(MSVC)
    target_compile_options(vividcode-launcher PRIVATE /utf-8 /W4)
    set_property(TARGET vividcode-launcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_dependencies(vividcode vividcode-viewer vividcode-launcher)
set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT vividcode-viewer)
enable_testing()
add_test(NAME pi_agent COMMAND vividcode-viewer --pi-test -platform offscreen)
set_tests_properties(pi_agent PROPERTIES TIMEOUT 90)
add_test(NAME ai_explanations COMMAND vividcode-viewer --ai-test -platform offscreen)
set_tests_properties(ai_explanations PROPERTIES TIMEOUT 90)
find_package(Python3 3.12 QUIET COMPONENTS Interpreter)
add_test(NAME prototype_smoke COMMAND vividcode-viewer --self-test -platform offscreen)
set_tests_properties(prototype_smoke PROPERTIES TIMEOUT 30)
add_test(NAME cli_lifecycle COMMAND powershell -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.Tests.ps1" -RuntimeDirectory "$<TARGET_FILE_DIR:vividcode>")
set_tests_properties(cli_lifecycle PROPERTIES TIMEOUT 300)
if(Python3_Interpreter_FOUND)
add_test(NAME generation_commands COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_generation_commands.py" -v)
set_tests_properties(generation_commands PROPERTIES ENVIRONMENT "VIVIDCODE_TEST_RUNTIME=$<TARGET_FILE_DIR:vividcode>" TIMEOUT 120)
add_test(NAME semantic_backend COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_analyzer.py" -v)
set_tests_properties(semantic_backend PROPERTIES TIMEOUT 120)
endif()
add_test(NAME semantic_ui COMMAND powershell -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/shared-viewer.Tests.ps1" -RuntimeDirectory "$<TARGET_FILE_DIR:vividcode>")
set_tests_properties(semantic_ui PROPERTIES TIMEOUT 120)
if(Python3_Interpreter_FOUND)
add_test(NAME csharp_backend COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_csharp_analyzer.py" -v)
set_tests_properties(csharp_backend PROPERTIES ENVIRONMENT "VIVIDCODE_CSHARP_ANALYZER=$<TARGET_FILE_DIR:vividcode>/csharp-analyzer/vividcode-csharp.exe" TIMEOUT 300)
endif()

add_test(NAME csharp_ui COMMAND powershell -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/shared-viewer.Tests.ps1" -RuntimeDirectory "$<TARGET_FILE_DIR:vividcode>" -CSharp)
set_tests_properties(csharp_ui PROPERTIES TIMEOUT 150)
if(Python3_Interpreter_FOUND)
add_test(NAME python_backend COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_python_indexer.py" -v)
set_tests_properties(python_backend PROPERTIES TIMEOUT 90)
endif()

    -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/shared-viewer.Tests.ps1" -RuntimeDirectory "$<TARGET_FILE_DIR:vividcode>" -Python)

if(Python3_Interpreter_FOUND)
add_test(NAME javascript_backend COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_javascript_indexer.py" -v)

set_tests_properties(java_backend PROPERTIES ENVIRONMENT "VIVIDCODE_JAVA_RUNTIME=$<TARGET_FILE_DIR:vividcode>" TIMEOUT 240)
