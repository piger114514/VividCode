cmake_minimum_required(VERSION 3.21)
project(vividcode LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_library(vividcode_store STATIC ProjectStore.cpp ProjectStore.h ProjectScanner.h)
target_link_libraries(vividcode_store PUBLIC Qt6::Core)
add_executable(vividcode cli_main.cpp)
target_link_libraries(vividcode PRIVATE vividcode_store)
add_executable(vividcode-viewer WIN32 main.cpp VividCode.cpp VividCode.h
    SemanticPanel.cpp SemanticPanel.h
    RelationGraph.cpp RelationGraph.h
    ProjectScanner.h VividCode.ui VividCode.qrc)
target_link_libraries(vividcode-viewer PRIVATE Qt6::Widgets vividcode_store)
if(MSVC)
    foreach(target vividcode vividcode-viewer vividcode_store)
        target_compile_options(${target} PRIVATE /utf-8 /W4)
    endforeach()
endif()
add_custom_command(TARGET vividcode-viewer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/demo" "$<TARGET_FILE_DIR:vividcode-viewer>/demo")
set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT vividcode-viewer)
enable_testing()
add_test(NAME prototype_smoke COMMAND vividcode-viewer --self-test -platform offscreen)
set_tests_properties(prototype_smoke PROPERTIES TIMEOUT 30)
add_test(NAME cli_lifecycle COMMAND powershell -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.Tests.ps1" -RuntimeDirectory "$<TARGET_FILE_DIR:vividcode>")
set_tests_properties(cli_lifecycle PROPERTIES TIMEOUT 300)
add_test(NAME semantic_backend COMMAND "$<TARGET_FILE_DIR:vividcode>/analyzer/python/python.exe" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_analyzer.py" -v)
set_tests_properties(semantic_backend PROPERTIES ENVIRONMENT "VIVIDCODE_LIBCLANG=$<TARGET_FILE_DIR:vividcode>/analyzer/lib/libclang.dll" TIMEOUT 120)
add_test(NAME semantic_ui COMMAND vividcode-viewer --semantic-test -platform offscreen)
set_tests_properties(semantic_ui PROPERTIES TIMEOUT 120)
